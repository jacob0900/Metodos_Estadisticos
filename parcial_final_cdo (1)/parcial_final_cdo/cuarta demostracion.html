\[ \mathrm{MSR}=\frac{\mathrm{SSR}}{k} \]
\[ \frac{\mathrm{SSR}}{\sigma^{2}}\sim \chi^{2}_{k}(\lambda) \]
\[ \lambda=\frac{\beta'X'X\beta}{\sigma^{2}} \]
\[ E\!\left[\frac{\mathrm{SSR}}{\sigma^{2}}\right]=k+\lambda \]
\[ E(\mathrm{SSR})=\sigma^{2}(k+\lambda)=\sigma^{2}k+\beta'X'X\beta \]
\[ E(\mathrm{MSR})=\frac{E(\mathrm{SSR})}{k} =\frac{\sigma^{2}k+\beta'X'X\beta}{k} =\sigma^{2}+\frac{\beta'X'X\beta}{k} \]
