\[ E(MSE) = \sigma^2 \]
\[ y = X\beta + \varepsilon,\qquad E(\varepsilon)=0,\qquad Var(\varepsilon)=\sigma^2 I \]
\[ H = X(X'X)^{-1}X' \]
\[ e = y - \hat{y} = (I - H)y \]
\[ \begin{aligned} e &= (I-H)(X\beta + \varepsilon)\\ &= (I-H)X\beta + (I-H)\varepsilon\\ &= (I-H)\varepsilon \end{aligned} \]
\[ SSE = e'(I-H)e \]
\[ SSE = \varepsilon'(I-H)\varepsilon \]
\[ E(SSE) = tr\left[(I-H)Var(\varepsilon)\right] \]
\[ Var(\varepsilon) = \sigma^2 I \]
\[ E(SSE) = \sigma^2\, tr(I-H) \]
\[ tr(I-H) = n - tr(H) \]
\[ tr(H) = rank(H) = p \]
\[ E(SSE) = \sigma^2 (n - p) \]
\[ MSE = \frac{SSE}{n-p} \]
\[ E(MSE) = \frac{E(SSE)}{n-p} = \frac{\sigma^2(n-p)}{n-p} = \sigma^2 \]
