\[ \hat{\beta} = (X'X)^{-1}X'y \]
\[ \begin{aligned} E(\hat{\beta}) &= E\big[(X'X)^{-1}X'y\big] \\ &= E\big[(X'X)^{-1}X'(X\beta + \varepsilon)\big] \\ &= E\big[(X'X)^{-1}X'X\beta + (X'X)^{-1}X'\varepsilon\big] \\ &= E\big[\beta + (X'X)^{-1}X'\varepsilon\big] \\ &= \beta + (X'X)^{-1}X' E(\varepsilon) \end{aligned} \]
\[ E(\varepsilon) = 0 \]
\[ E(\hat{\beta}) = \beta \]
